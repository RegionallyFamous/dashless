function dl_probe_snapshot($dir,$owner) {
    wp_set_current_user($owner);
    $before=get_option('dashless_content_version');
    $collections=array();
    foreach(array('posts','pages','media','categories','tags') as $type) {
        $all=array();$page=1;
        do {
            $request=new WP_REST_Request('GET','/wp/v2/'.$type);
            $query=array('per_page'=>100,'page'=>$page,'context'=>'view');
            if(in_array($type,array('posts','pages'),true))$query['status']='publish';
            if($type==='posts'){$query['orderby']='date';$query['order']='desc';}
            if($type==='pages'){$query['orderby']='menu_order';$query['order']='asc';}
            if(in_array($type,array('categories','tags'),true)){$query['hide_empty']=false;$query['orderby']='name';$query['order']='asc';}
            $request->set_query_params($query);$response=rest_do_request($request);
            if($response->is_error())throw new Exception('Snapshot read failed: '.$type);
            $all=array_merge($all,$response->get_data());$headers=$response->get_headers();
            $pages=(int)($headers['X-WP-TotalPages']??1);$page++;
        } while($page<=$pages);
        $collections[$type]=$all;
    }
    $after=get_option('dashless_content_version');
    if($before!==$after)throw new Exception('Content changed during snapshot');
    $site=rest_do_request(new WP_REST_Request('GET','/dashless/v1/site'))->get_data();
    $snapshot=array('generation'=>$after,'collections'=>$collections,'site'=>$site);
    file_put_contents($dir.'/snapshot.json',wp_json_encode($snapshot));chmod($dir.'/snapshot.json',0600);
    return array('generation'=>$after,'counts'=>array_map('count',$collections));
}
