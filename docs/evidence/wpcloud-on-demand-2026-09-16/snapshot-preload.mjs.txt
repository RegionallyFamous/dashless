import fs from 'node:fs/promises';
import path from 'node:path';
import { createHash } from 'node:crypto';
const originalFetch=globalThis.fetch;
const snapshot=JSON.parse(await fs.readFile(process.env.PROBE_SNAPSHOT,'utf8'));
globalThis.fetch=async (input,init)=>{
    const url=new URL(input instanceof Request?input.url:String(input));
    if(url.origin!=='https://teddy.blog')return originalFetch(input,init);
    if(url.pathname.startsWith('/wp-json/')) {
        const route=url.pathname.slice('/wp-json/'.length);
        let data;
        const single=route.match(/^wp\/v2\/media\/(\d+)$/);
        if(single) data=snapshot.collections.media.find(m=>m.id===Number(single[1]));
        else if(route==='dashless/v1/site')data=snapshot.site;
        else {
            const match=route.match(/^wp\/v2\/(posts|pages|categories|tags)$/);
            if(!match)throw new Error('Unmapped snapshot route '+route);
            let all=snapshot.collections[match[1]];
            if(url.searchParams.get('hide_empty')==='true')all=all.filter(x=>x.count>0);
            const size=Number(url.searchParams.get('per_page')||100),page=Number(url.searchParams.get('page')||1);
            data=all.slice((page-1)*size,page*size);
            return new Response(JSON.stringify(data),{headers:{'Content-Type':'application/json','X-WP-TotalPages':String(Math.ceil(all.length/size)),'X-WP-Total':String(all.length)}});
        }
        if(!data)throw new Error('Snapshot missing '+route);
        return new Response(JSON.stringify(data),{headers:{'Content-Type':'application/json'}});
    }
    if(url.pathname.startsWith('/wp-content/uploads/')) {
        const base='/srv/htdocs/wp-content/uploads';
        const file=path.resolve(base,decodeURIComponent(url.pathname.slice('/wp-content/uploads/'.length)));
        if(!file.startsWith(base+'/'))throw new Error('Invalid media path');
        const bytes=await fs.readFile(file);
        const etag='"'+createHash('sha256').update(bytes).digest('hex')+'"';
        if(new Headers(init?.headers).get('If-None-Match')===etag)return new Response(null,{status:304});
        return new Response(bytes,{headers:{'ETag':etag,'Content-Length':String(bytes.length)}});
    }
    return originalFetch(input,init);
};
